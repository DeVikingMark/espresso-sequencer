50async fn handle_drb_result<TYPES: NodeType, I: NodeImplementation<TYPES>>(
51    membership: &EpochMembership<TYPES>,
52    storage: &Arc<RwLock<I::Storage>>,
53    drb_result: DrbResult,
54) {
55    tracing::debug!("Calling add_drb_result for epoch {:?}", membership.epoch());
56
57    // membership.epoch should always be Some
58    if let Some(epoch) = membership.epoch() {
59        if let Err(e) = storage
60            .write()
61            .await
62            .add_drb_result(epoch, drb_result)
63            .await
64        {
65            tracing::error!("Failed to store drb result for epoch {:?}: {}", epoch, e);
66        }
67    }
68
69    membership.add_drb_result(drb_result).await;
70}

72/// Store the DRB result from the computation task to the shared `results` table.
73///
74/// Returns the result if it exists.
75async fn store_and_get_computed_drb_result<
76    TYPES: NodeType,
77    I: NodeImplementation<TYPES>,
78    V: Versions,
79>(
80    epoch_number: TYPES::Epoch,
81    task_state: &mut QuorumVoteTaskState<TYPES, I, V>,
82) -> Result<DrbResult> {
83    // Return the result if it's already in the table.
84    if let Some(computed_result) = task_state
85        .consensus
86        .read()
87        .await
88        .drb_seeds_and_results
89        .results
90        .get(&epoch_number)
91    {
92        return Ok(*computed_result);
93    }
94
95    let (task_epoch, computation) =
96        (&mut task_state.drb_computation).context(warn!("DRB computation task doesn't exist."))?;
97
98    ensure!(
99        *task_epoch == epoch_number,
100        info!("DRB computation is not for the next epoch.")
101    );
102
103    ensure!(
104        computation.is_finished(),
105        info!("DRB computation has not yet finished.")
106    );
107
108    match computation.await {
109        Ok(result) => {
110            let mut consensus_writer = task_state.consensus.write().await;
111            consensus_writer
112                .drb_seeds_and_results
113                .results
114                .insert(epoch_number, result);
115            drop(consensus_writer);
116
117            handle_drb_result::<TYPES, I>(
118                &task_state
119                    .membership
120                    .membership_for_epoch(Some(epoch_number))
121                    .await?,
122                &task_state.storage,
123                result,
124            )
125            .await;
126            task_state.drb_computation = None;
127            Ok(result)
128        },
129        Err(e) => Err(warn!("Error in DRB calculation: {:?}.", e)),
130    }
131}

133/// Verify the DRB result from the proposal for the next epoch if this is the last block of the
134/// current epoch.
135///
136/// Uses the result from `start_drb_task`.
137///
138/// Returns an error if we should not vote.
139async fn verify_drb_result<TYPES: NodeType, I: NodeImplementation<TYPES>, V: Versions>(
140    proposal: &QuorumProposalWrapper<TYPES>,
141    task_state: &mut QuorumVoteTaskState<TYPES, I, V>,
142) -> Result<()> {
143    // Skip if this is not the expected block.
144    if task_state.epoch_height == 0
145        || !is_last_block_in_epoch(
146            proposal.block_header().block_number(),
147            task_state.epoch_height,
148        )
149    {
150        tracing::debug!("Skipping DRB result verification");
151        return Ok(());
152    }
153
154    // #3967 REVIEW NOTE: Check if this is the right way to decide if we're doing epochs
155    // Alternatively, should we just return Err() if epochs aren't happening here? Or can we assume
156    // that epochs are definitely happening by virtue of getting here?
157    let epoch = option_epoch_from_block_number::<TYPES>(
158        task_state
159            .upgrade_lock
160            .epochs_enabled(proposal.view_number())
161            .await,
162        proposal.block_header().block_number(),
163        task_state.epoch_height,
164    );
165
166    let proposal_result = proposal
167        .next_drb_result()
168        .context(info!("Proposal is missing the DRB result."))?;
169
170    if let Some(epoch_val) = epoch {
171        let has_stake_current_epoch = task_state
172            .membership
173            .membership_for_epoch(epoch)
174            .await
175            .context(warn!("No stake table for epoch"))?
176            .has_stake(&task_state.public_key)
177            .await;
178
179        if has_stake_current_epoch {
180            let computed_result =
181                store_and_get_computed_drb_result(epoch_val + 1, task_state).await?;
182
183            ensure!(proposal_result == computed_result, warn!("Our calculated DRB result is {:?}, which does not match the proposed DRB result of {:?}", computed_result, proposal_result));
184        }
185
186        Ok(())
187    } else {
188        Err(error!("Epochs are not available"))
189    }
190}

192/// Start the DRB computation task for the next epoch.
193///
194/// Uses the seed previously stored in `store_drb_seed_and_result`.
195async fn start_drb_task<TYPES: NodeType, I: NodeImplementation<TYPES>, V: Versions>(
196    proposal: &QuorumProposalWrapper<TYPES>,
197    task_state: &mut QuorumVoteTaskState<TYPES, I, V>,
198) {
199    // #3967 REVIEW NOTE: Should we just exit early if we aren't doing epochs?
200    if proposal.epoch().is_none() {
201        return;
202    }
203
204    let current_epoch_number = TYPES::Epoch::new(epoch_from_block_number(
205        proposal.block_header().block_number(),
206        task_state.epoch_height,
207    ));
208
209    let Ok(epoch_membership) = task_state
210        .membership
211        .membership_for_epoch(Some(current_epoch_number))
212        .await
213    else {
214        tracing::warn!("No Stake Table for Epoch = {:?}", current_epoch_number);
215        return;
216    };
217
218    // Start the new task if we're in the committee for this epoch
219    if epoch_membership.has_stake(&task_state.public_key).await {
220        let new_epoch_number = current_epoch_number + 1;
221
222        // If a task is currently live AND has finished, join it and save the result.
223        // If the epoch for the calculation was the same as the provided epoch, return.
224        // If a task is currently live and NOT finished, abort it UNLESS the task epoch is the
225        // same as cur_epoch, in which case keep letting it run and return.
226        // Continue the function if a task should be spawned for the given epoch.
227        if let Some((task_epoch, join_handle)) = &mut task_state.drb_computation {
228            if join_handle.is_finished() {
229                match join_handle.await {
230                    Ok(result) => {
231                        task_state
232                            .consensus
233                            .write()
234                            .await
235                            .drb_seeds_and_results
236                            .results
237                            .insert(*task_epoch, result);
238                        handle_drb_result::<TYPES, I>(
239                            &epoch_membership,
240                            &task_state.storage,
241                            result,
242                        )
243                        .await;
244                        task_state.drb_computation = None;
245                    },
246                    Err(e) => {
247                        tracing::error!("error joining DRB computation task: {e:?}");
248                    },
249                }
250            } else if *task_epoch == new_epoch_number {
251                return;
252            } else {
253                join_handle.abort();
254                task_state.drb_computation = None;
255            }
256        }
257
258        // In case we somehow ended up processing this epoch already, don't start it again
259        let mut consensus_writer = task_state.consensus.write().await;
260        if consensus_writer
261            .drb_seeds_and_results
262            .results
263            .contains_key(&new_epoch_number)
264        {
265            return;
266        }
267
268        if let Entry::Occupied(entry) = consensus_writer
269            .drb_seeds_and_results
270            .seeds
271            .entry(new_epoch_number)
272        {
273            let drb_seed_input = *entry.get();
274            let new_drb_task = spawn(async move { compute_drb_result::<TYPES>(drb_seed_input) });
275            task_state.drb_computation = Some((new_epoch_number, new_drb_task));
276            entry.remove();
277        }
278    }
279}

281/// Store the DRB seed two epochs in advance and the computed or received DRB result for next
282/// epoch.
283///
284/// We store a combination of the following data.
285/// * The DRB seed two epochs in advance, if the third from the last block, i.e., the epoch root,
286///     is decided and we are in the quorum committee of the next epoch.
287/// * The computed result for the next epoch, if the third from the last block is decided.
288/// * The received result for the next epoch, if the last block of the epoch is decided and we are
289///     in the quorum committee of the committee of the next epoch.
290///
291/// Special cases:
292/// * Epoch 0: No DRB computation since we'll transition to epoch 1 immediately.
293/// * Epoch 1 and 2: No computed DRB result since when we first start the computation in epoch 1,
294///     the result is for epoch 3.
295///
296/// We don't need to handle the special cases explicitly here, because the first leaf with which
297/// we'll start the DRB computation is for epoch 3.
298async fn store_drb_seed_and_result<TYPES: NodeType, I: NodeImplementation<TYPES>, V: Versions>(
299    task_state: &mut QuorumVoteTaskState<TYPES, I, V>,
300    decided_leaf: &Leaf2<TYPES>,
301) -> Result<()> {
302    if task_state.epoch_height == 0 {
303        tracing::info!("Epoch height is 0, skipping DRB storage.");
304        return Ok(());
305    }
306
307    let decided_block_number = decided_leaf.block_header().block_number();
308    let current_epoch_number = TYPES::Epoch::new(epoch_from_block_number(
309        decided_block_number,
310        task_state.epoch_height,
311    ));
312
313    // Skip storing the seed and computed result if this is not the epoch root.
314    if is_epoch_root(decided_block_number, task_state.epoch_height) {
315        // Cancel old DRB computation tasks.
316        let mut consensus_writer = task_state.consensus.write().await;
317        consensus_writer
318            .drb_seeds_and_results
319            .garbage_collect(current_epoch_number);
320        drop(consensus_writer);
321
322        // Store the DRB result for the next epoch, which will be used by the proposal task to
323        // include in the proposal in the last block of this epoch.
324        store_and_get_computed_drb_result(current_epoch_number + 1, task_state).await?;
325
326        // Store the DRB seed input for the epoch after the next one.
327        let Ok(drb_seed_input_vec) = bincode::serialize(&decided_leaf.justify_qc().signatures)
328        else {
329            bail!("Failed to serialize the QC signature.");
330        };
331
332        // TODO: Replace the leader election with a weighted version.
333        // <https://github.com/EspressoSystems/HotShot/issues/3898>
334        let mut drb_seed_input = [0u8; 32];
335        let len = drb_seed_input_vec.len().min(32);
336        drb_seed_input[..len].copy_from_slice(&drb_seed_input_vec[..len]);
337
338        task_state
339            .consensus
340            .write()
341            .await
342            .drb_seeds_and_results
343            .store_seed(current_epoch_number + 2, drb_seed_input);
344    }
345    // Skip storing the received result if this is not the last block.
346    else if is_last_block_in_epoch(decided_block_number, task_state.epoch_height) {
347        if let Some(result) = decided_leaf.next_drb_result {
348            // We don't need to check value existence and consistency because it should be
349            // impossible to decide on a block with different DRB results.
350            task_state
351                .consensus
352                .write()
353                .await
354                .drb_seeds_and_results
355                .results
356                .insert(current_epoch_number + 1, result);
357            handle_drb_result::<TYPES, I>(
358                &task_state
359                    .membership
360                    .membership_for_epoch(Some(current_epoch_number + 1))
361                    .await?,
362                &task_state.storage,
363                result,
364            )
365            .await;
366        } else {
367            bail!("The last block of the epoch is decided but doesn't contain a DRB result.");
368        }
369    }
370    Ok(())
371}

373/// Handles the `QuorumProposalValidated` event.
374#[instrument(skip_all, fields(id = task_state.id, view = *proposal.view_number()))]
375pub(crate) async fn handle_quorum_proposal_validated<
376    TYPES: NodeType,
377    I: NodeImplementation<TYPES>,
378    V: Versions,
379>(
380    proposal: &QuorumProposalWrapper<TYPES>,
381    task_state: &mut QuorumVoteTaskState<TYPES, I, V>,
382) -> Result<()> {
383    let version = task_state
384        .upgrade_lock
385        .version(proposal.view_number())
386        .await?;
387
388    if version >= V::Epochs::VERSION {
389        // Don't vote if the DRB result verification fails.
390        verify_drb_result(proposal, task_state).await?;
391        start_drb_task(proposal, task_state).await;
392    }
393
394    let LeafChainTraversalOutcome {
395        new_locked_view_number,
396        new_decided_view_number,
397        new_decide_qc,
398        leaf_views,
399        included_txns,
400        decided_upgrade_cert,
401    } = if version >= V::Epochs::VERSION {
402        decide_from_proposal_2::<TYPES, I>(
403            proposal,
404            OuterConsensus::new(Arc::clone(&task_state.consensus.inner_consensus)),
405            Arc::clone(&task_state.upgrade_lock.decided_upgrade_certificate),
406            &task_state.public_key,
407            version >= V::Epochs::VERSION,
408            task_state.membership.membership(),
409            &task_state.storage,
410        )
411        .await
412    } else {
413        decide_from_proposal::<TYPES, I, V>(
414            proposal,
415            OuterConsensus::new(Arc::clone(&task_state.consensus.inner_consensus)),
416            Arc::clone(&task_state.upgrade_lock.decided_upgrade_certificate),
417            &task_state.public_key,
418            version >= V::Epochs::VERSION,
419            task_state.membership.membership(),
420            &task_state.storage,
421        )
422        .await
423    };
424
425    if let Some(cert) = &task_state.staged_epoch_upgrade_certificate {
426        if leaf_views.last().unwrap().leaf.height() >= task_state.epoch_upgrade_block_height {
427            let mut decided_certificate_lock = task_state
428                .upgrade_lock
429                .decided_upgrade_certificate
430                .write()
431                .await;
432            *decided_certificate_lock = Some(cert.clone());
433            drop(decided_certificate_lock);
434
435            let _ = task_state
436                .storage
437                .write()
438                .await
439                .update_decided_upgrade_certificate(Some(cert.clone()))
440                .await;
441
442            task_state.staged_epoch_upgrade_certificate = None;
443        }
444    };
445
446    if let Some(cert) = decided_upgrade_cert.clone() {
447        if cert.data.new_version == V::Epochs::VERSION {
448            task_state.staged_epoch_upgrade_certificate = Some(cert);
449
450            let epoch_height = task_state.consensus.read().await.epoch_height;
451            let first_epoch_number = TYPES::Epoch::new(epoch_from_block_number(
452                task_state.epoch_upgrade_block_height,
453                epoch_height,
454            ));
455            tracing::debug!("Calling set_first_epoch for epoch {:?}", first_epoch_number);
456            task_state
457                .membership
458                .membership()
459                .write()
460                .await
461                .set_first_epoch(first_epoch_number, INITIAL_DRB_RESULT);
462        } else {
463            let mut decided_certificate_lock = task_state
464                .upgrade_lock
465                .decided_upgrade_certificate
466                .write()
467                .await;
468            *decided_certificate_lock = Some(cert.clone());
469            drop(decided_certificate_lock);
470
471            let _ = task_state
472                .storage
473                .write()
474                .await
475                .update_decided_upgrade_certificate(Some(cert.clone()))
476                .await;
477        }
478    }
479
480    let mut consensus_writer = task_state.consensus.write().await;
481    if let Some(locked_view_number) = new_locked_view_number {
482        consensus_writer.update_locked_view(locked_view_number)?;
483    }
484
485    #[allow(clippy::cast_precision_loss)]
486    if let Some(decided_view_number) = new_decided_view_number {
487        // Bring in the cleanup crew. When a new decide is indeed valid, we need to clear out old memory.
488
489        let old_decided_view = consensus_writer.last_decided_view();
490        consensus_writer.collect_garbage(old_decided_view, decided_view_number);
491
492        // Set the new decided view.
493        consensus_writer.update_last_decided_view(decided_view_number)?;
494
495        consensus_writer
496            .metrics
497            .last_decided_time
498            .set(Utc::now().timestamp().try_into().unwrap());
499        consensus_writer.metrics.invalid_qc.set(0);
500        consensus_writer
501            .metrics
502            .last_decided_view
503            .set(usize::try_from(consensus_writer.last_decided_view().u64()).unwrap());
504        let cur_number_of_views_per_decide_event =
505            *proposal.view_number() - consensus_writer.last_decided_view().u64();
506        consensus_writer
507            .metrics
508            .number_of_views_per_decide_event
509            .add_point(cur_number_of_views_per_decide_event as f64);
510
511        tracing::debug!(
512            "Sending Decide for view {:?}",
513            consensus_writer.last_decided_view()
514        );
515
516        // We don't need to hold this while we broadcast
517        drop(consensus_writer);
518
519        // Send an update to everyone saying that we've reached a decide
520        broadcast_event(
521            Event {
522                view_number: decided_view_number,
523                event: EventType::Decide {
524                    leaf_chain: Arc::new(leaf_views.clone()),
525                    // This is never none if we've reached a new decide, so this is safe to unwrap.
526                    qc: Arc::new(new_decide_qc.unwrap()),
527                    block_size: included_txns.map(|txns| txns.len().try_into().unwrap()),
528                },
529            },
530            &task_state.output_event_stream,
531        )
532        .await;
533        tracing::debug!("Successfully sent decide event");
534
535        if version >= V::Epochs::VERSION {
536            // `leaf_views.last()` is never none if we've reached a new decide, so this is safe to
537            // unwrap.
538            store_drb_seed_and_result(task_state, &leaf_views.last().unwrap().leaf).await?;
539        }
540    }
541
542    Ok(())
543}

545/// Updates the shared consensus state with the new voting data.
546#[instrument(skip_all, target = "VoteDependencyHandle", fields(view = *view_number))]
547#[allow(clippy::too_many_arguments)]
548pub(crate) async fn update_shared_state<
549    TYPES: NodeType,
550    I: NodeImplementation<TYPES>,
551    V: Versions,
552>(
553    consensus: OuterConsensus<TYPES>,
554    sender: Sender<Arc<HotShotEvent<TYPES>>>,
555    receiver: InactiveReceiver<Arc<HotShotEvent<TYPES>>>,
556    membership: EpochMembershipCoordinator<TYPES>,
557    public_key: TYPES::SignatureKey,
558    private_key: <TYPES::SignatureKey as SignatureKey>::PrivateKey,
559    upgrade_lock: UpgradeLock<TYPES, V>,
560    view_number: TYPES::View,
561    instance_state: Arc<TYPES::InstanceState>,
562    storage: Arc<RwLock<I::Storage>>,
563    proposed_leaf: &Leaf2<TYPES>,
564    vid_share: &Proposal<TYPES, VidDisperseShare<TYPES>>,
565    parent_view_number: Option<TYPES::View>,
566    epoch_height: u64,
567) -> Result<()> {
568    let justify_qc = &proposed_leaf.justify_qc();
569
570    let consensus_reader = consensus.read().await;
571    // Try to find the validated view within the validated state map. This will be present
572    // if we have the saved leaf, but if not we'll get it when we fetch_proposal.
573    let mut maybe_validated_view = parent_view_number.and_then(|view_number| {
574        consensus_reader
575            .validated_state_map()
576            .get(&view_number)
577            .cloned()
578    });
579
580    // Justify qc's leaf commitment should be the same as the parent's leaf commitment.
581    let mut maybe_parent = consensus_reader
582        .saved_leaves()
583        .get(&justify_qc.data.leaf_commit)
584        .cloned();
585
586    drop(consensus_reader);
587
588    maybe_parent = match maybe_parent {
589        Some(p) => Some(p),
590        None => {
591            match fetch_proposal(
592                justify_qc.view_number(),
593                sender.clone(),
594                receiver.activate_cloned(),
595                membership.clone(),
596                OuterConsensus::new(Arc::clone(&consensus.inner_consensus)),
597                public_key.clone(),
598                private_key.clone(),
599                &upgrade_lock,
600                epoch_height,
601            )
602            .await
603            .ok()
604            {
605                Some((leaf, view)) => {
606                    maybe_validated_view = Some(view);
607                    Some(leaf)
608                },
609                None => None,
610            }
611        },
612    };
613
614    let parent = maybe_parent.context(info!(
615        "Proposal's parent missing from storage with commitment: {:?}, proposal view {:?}",
616        justify_qc.data.leaf_commit,
617        proposed_leaf.view_number(),
618    ))?;
619
620    let Some(validated_view) = maybe_validated_view else {
621        bail!(
622            "Failed to fetch view for parent, parent view {:?}",
623            parent_view_number
624        );
625    };
626
627    let (Some(parent_state), maybe_parent_delta) = validated_view.state_and_delta() else {
628        bail!("Parent state not found! Consensus internally inconsistent");
629    };
630
631    let (state, delta) = if is_last_block_in_epoch(proposed_leaf.height(), epoch_height)
632        && proposed_leaf.height() == parent.height()
633        && maybe_parent_delta.is_some()
634    {
635        // This is an epoch transition. We do not want to call `validate_and_apply_header` second
636        // time for the same block. Just grab the state and delta from the parent and update the shared
637        // state with those.
638        (parent_state, maybe_parent_delta.unwrap())
639    } else {
640        let version = upgrade_lock.version(view_number).await?;
641
642        let (validated_state, state_delta) = parent_state
643            .validate_and_apply_header(
644                &instance_state,
645                &parent,
646                &proposed_leaf.block_header().clone(),
647                vid_share.data.payload_byte_len(),
648                version,
649                *view_number,
650            )
651            .await
652            .wrap()
653            .context(warn!("Block header doesn't extend the proposal!"))?;
654
655        (Arc::new(validated_state), Arc::new(state_delta))
656    };
657
658    // Now that we've rounded everyone up, we need to update the shared state
659    let mut consensus_writer = consensus.write().await;
660
661    if let Err(e) = consensus_writer.update_leaf(
662        proposed_leaf.clone(),
663        Arc::clone(&state),
664        Some(Arc::clone(&delta)),
665    ) {
666        tracing::trace!("{e:?}");
667    }
668
669    // Kick back our updated structures for downstream usage.
670    let new_leaves = consensus_writer.saved_leaves().clone();
671    let new_state = consensus_writer.validated_state_map().clone();
672    drop(consensus_writer);
673
674    // Send the new state up to the sequencer.
675    storage
676        .write()
677        .await
678        .update_undecided_state2(new_leaves, new_state)
679        .await
680        .wrap()
681        .context(error!("Failed to update undecided state"))?;
682
683    Ok(())
684}

686/// Submits the `QuorumVoteSend` event if all the dependencies are met.
687#[instrument(skip_all, fields(name = "Submit quorum vote", level = "error"))]
688#[allow(clippy::too_many_arguments)]
689pub(crate) async fn submit_vote<TYPES: NodeType, I: NodeImplementation<TYPES>, V: Versions>(
690    sender: Sender<Arc<HotShotEvent<TYPES>>>,
691    membership: EpochMembership<TYPES>,
692    public_key: TYPES::SignatureKey,
693    private_key: <TYPES::SignatureKey as SignatureKey>::PrivateKey,
694    upgrade_lock: UpgradeLock<TYPES, V>,
695    view_number: TYPES::View,
696    storage: Arc<RwLock<I::Storage>>,
697    leaf: Leaf2<TYPES>,
698    vid_share: Proposal<TYPES, VidDisperseShare<TYPES>>,
699    extended_vote: bool,
700    epoch_height: u64,
701) -> Result<()> {
702    let committee_member_in_current_epoch = membership.has_stake(&public_key).await;
703    // If the proposed leaf is for the last block in the epoch and the node is part of the quorum committee
704    // in the next epoch, the node should vote to achieve the double quorum.
705    let committee_member_in_next_epoch = leaf.with_epoch
706        && is_last_block_in_epoch(leaf.height(), epoch_height)
707        && membership.next_epoch().await?.has_stake(&public_key).await;
708
709    ensure!(
710        committee_member_in_current_epoch || committee_member_in_next_epoch,
711        info!(
712            "We were not chosen for quorum committee on {:?}",
713            view_number
714        )
715    );
716
717    // Create and send the vote.
718    let vote = QuorumVote2::<TYPES>::create_signed_vote(
719        QuorumData2 {
720            leaf_commit: leaf.commit(),
721            epoch: membership.epoch(),
722        },
723        view_number,
724        &public_key,
725        &private_key,
726        &upgrade_lock,
727    )
728    .await
729    .wrap()
730    .context(error!("Failed to sign vote. This should never happen."))?;
731    // Add to the storage.
732    storage
733        .write()
734        .await
735        .append_vid_general(&vid_share)
736        .await
737        .wrap()
738        .context(error!("Failed to store VID share"))?;
739
740    if extended_vote {
741        tracing::debug!("sending extended vote to everybody",);
742        broadcast_event(
743            Arc::new(HotShotEvent::ExtendedQuorumVoteSend(vote)),
744            &sender,
745        )
746        .await;
747    } else {
748        tracing::debug!(
749            "sending vote to next quorum leader {:?}",
750            vote.view_number() + 1
751        );
752        broadcast_event(Arc::new(HotShotEvent::QuorumVoteSend(vote)), &sender).await;
753    }
754
755    Ok(())
756}
