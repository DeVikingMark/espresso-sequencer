1//! Sequencer node persistence.
2//!
3//! This module implements the persistence required for a sequencer node to rejoin the network and
4//! resume participating in consensus, in the event that its process crashes or is killed and loses
5//! all in-memory state.
6//!
7//! This is distinct from the query service persistent storage found in the `api` module, which is
8//! an extension that node operators can opt into. This module defines the minimum level of
9//! persistence which is _required_ to run a node.
10

87    #[derive(Clone, Debug, Default)]
88    struct EventCollector {
89        events: Arc<RwLock<Vec<Event>>>,
90    }
91
92    impl EventCollector {
93        async fn leaf_chain(&self) -> Vec<LeafInfo<SeqTypes>> {
94            self.events
95                .read()
96                .await
97                .iter()
98                .flat_map(|event| {
99                    let EventType::Decide { leaf_chain, .. } = &event.event else {
100                        panic!("expected decide event, got {event:?}");
101                    };
102                    leaf_chain.iter().cloned().rev()
103                })
104                .collect::<Vec<_>>()
105        }
106    }
107
108    #[async_trait]
109    impl EventConsumer for EventCollector {
110        async fn handle_event(&self, event: &Event) -> anyhow::Result<()> {
111            self.events.write().await.push(event.clone());
112            Ok(())
113        }
114    }

451
452        // Test decide and garbage collection. Pass in a leaf chain with no VID shares or payloads,
453        // so we have to fetch the missing data from storage.
454        let leaves = [
455            Leaf2::from_quorum_proposal(&quorum_proposal1.data),
456            Leaf2::from_quorum_proposal(&quorum_proposal2.data),
457            Leaf2::from_quorum_proposal(&quorum_proposal3.data),
458            Leaf2::from_quorum_proposal(&quorum_proposal4.data),
459        ];
460        let mut final_qc = leaves[3].justify_qc();
461        final_qc.view_number += 1;
462        final_qc.data.leaf_commit = Committable::commit(&leaf);
463        let qcs = [
464            leaves[1].justify_qc(),
465            leaves[2].justify_qc(),
466            leaves[3].justify_qc(),
467            final_qc,

470        assert_eq!(
471            storage.load_anchor_view().await.unwrap(),
472            ViewNumber::genesis()
473        );
474
475        let consumer = EventCollector::default();
476        let leaf_chain = leaves
477            .iter()
478            .take(3)
479            .map(|leaf| leaf_info(leaf.clone()))

481            .collect::<Vec<_>>();
482        tracing::info!(?leaf_chain, "decide view 2");
483        storage
484            .append_decided_leaves(
485                ViewNumber::new(2),
486                leaf_chain.iter().map(|(leaf, qc)| (leaf, (*qc).clone())),
487                &consumer,
488            )
489            .await
490            .unwrap();
491        assert_eq!(
492            storage.load_anchor_view().await.unwrap(),
493            ViewNumber::new(2)
494        );
495
496        for i in 0..=2 {
497            assert_eq!(
498                storage.load_da_proposal(ViewNumber::new(i)).await.unwrap(),
499                None
500            );
501
502            assert_eq!(
503                storage.load_vid_share(ViewNumber::new(i)).await.unwrap(),
504                None
505            );
506        }
507
508        assert_eq!(
509            storage.load_da_proposal(ViewNumber::new(3)).await.unwrap(),
510            Some(da_proposal3)
511        );
512
513        assert_eq!(
514            storage.load_vid_share(ViewNumber::new(3)).await.unwrap(),
515            Some(convert_proposal(vid_share3.clone()))
516        );
517
518        let proposals = storage.load_quorum_proposals().await.unwrap();
519        assert_eq!(
520            proposals,
521            BTreeMap::from_iter([(ViewNumber::new(3), quorum_proposal4)])
522        );
523
524        // A decide event should have been processed.
525        for (leaf, info) in leaves.iter().zip(consumer.leaf_chain().await.iter()) {
526            assert_eq!(info.leaf, *leaf);
527            let decided_vid_share = info.vid_share.as_ref().unwrap();
528            let view_number = match decided_vid_share {
529                VidDisperseShare::V0(share) => share.view_number,
530                VidDisperseShare::V1(share) => share.view_number,
531            };
532            assert_eq!(view_number, leaf.view_number());
533        }
534
535        // The decided leaf should not have been garbage collected.
536        assert_eq!(
537            storage.load_anchor_leaf().await.unwrap(),
538            Some((leaves[2].clone(), qcs[2].clone()))
539        );
540        assert_eq!(
541            storage.load_anchor_view().await.unwrap(),
542            leaves[2].view_number()
543        );
544
545        // Process a second decide event.
546        let consumer = EventCollector::default();
547        tracing::info!(leaf = ?leaves[3], qc = ?qcs[3], "decide view 3");
548        storage
549            .append_decided_leaves(
550                ViewNumber::new(3),
551                vec![(&leaf_info(leaves[3].clone()), qcs[3].clone())],
552                &consumer,
553            )
554            .await
555            .unwrap();
556        assert_eq!(
557            storage.load_anchor_view().await.unwrap(),
558            ViewNumber::new(3)
559        );
560
561        // A decide event should have been processed.
562        let events = consumer.events.read().await;
563        assert_eq!(events.len(), 1);
564        assert_eq!(events[0].view_number, ViewNumber::new(3));
565        let EventType::Decide { qc, leaf_chain, .. } = &events[0].event else {
566            panic!("expected decide event, got {:?}", events[0]);
567        };
568        assert_eq!(**qc, qcs[3]);
569        assert_eq!(leaf_chain.len(), 1);
570        let info = &leaf_chain[0];
571        assert_eq!(info.leaf, leaves[3]);
572
573        // The remaining data should have been GCed.
574        assert_eq!(
575            storage.load_da_proposal(ViewNumber::new(3)).await.unwrap(),
576            None
577        );
578
579        assert_eq!(
580            storage.load_vid_share(ViewNumber::new(3)).await.unwrap(),
581            None
582        );
583        assert_eq!(
584            storage.load_quorum_proposals().await.unwrap(),
585            BTreeMap::new()
586        );
587    }
