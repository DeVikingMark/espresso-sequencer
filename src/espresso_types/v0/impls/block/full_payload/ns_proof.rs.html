14    /// Returns the payload bytes for the `index`th namespace, along with a
15    /// proof of correctness for those bytes. Returns `None` on error.
16    ///
17    /// The namespace payload [`NsPayloadOwned`] is included as a hidden field
18    /// in the returned [`NsProof`]. A conventional API would instead return
19    /// `(NsPayload, NsProof)` and [`NsProof`] would not contain the namespace
20    /// payload.
21    /// ([`TxProof::new`](crate::block::namespace_payload::TxProof::new)
22    /// conforms to this convention.) In the future we should change this API to
23    /// conform to convention. But that would require a change to our RPC
24    /// endpoint API at [`endpoints`](crate::api::endpoints), which is a hassle.
25    pub fn new(payload: &Payload, index: &NsIndex, common: &ADVZCommon) -> Option<NsProof> {
26        let payload_byte_len = payload.byte_len();
27        if !payload_byte_len.is_consistent(common) {
28            tracing::warn!(
29                "payload byte len {} inconsistent with common {}",
30                payload_byte_len,
31                ADVZScheme::get_payload_byte_len(common)
32            );
33            return None; // error: payload byte len inconsistent with common
34        }
35        if !payload.ns_table().in_bounds(index) {
36            tracing::warn!("ns_index {:?} out of bounds", index);
37            return None; // error: index out of bounds
38        }
39        let ns_payload_range = payload.ns_table().ns_range(index, &payload_byte_len);
40
41        // TODO vid_scheme() arg should be u32 to match get_num_storage_nodes
42        // https://github.com/EspressoSystems/HotShot/issues/3298
43        let vid = advz_scheme(
44            ADVZScheme::get_num_storage_nodes(common).try_into().ok()?, // error: failure to convert u32 to usize
45        );
46
47        let ns_proof = if ns_payload_range.as_block_range().is_empty() {
48            None
49        } else {
50            Some(
51                vid.payload_proof(payload.encode(), ns_payload_range.as_block_range())
52                    .ok()?, // error: internal to payload_proof()
53            )
54        };
55
56        Some(NsProof {
57            ns_index: index.clone(),
58            ns_payload: payload.read_ns_payload(&ns_payload_range).to_owned(),
59            ns_proof,
60        })
61    }

63    /// Verify a [`NsProof`] against a payload commitment. Returns `None` on
64    /// error or if verification fails.
65    ///
66    /// There is no [`NsPayload`](crate::block::namespace_payload::NsPayload)
67    /// arg because this data is already included in the [`NsProof`]. See
68    /// [`NsProof::new`] for discussion.
69    ///
70    /// If verification is successful then return `(Vec<Transaction>,
71    /// NamespaceId)` obtained by post-processing the underlying
72    /// [`NsPayload`](crate::block::namespace_payload::NsPayload). Why? This
73    /// method might be run by a client in a WASM environment who might be
74    /// running non-Rust code, in which case the client is unable to perform
75    /// this post-processing himself.
76    pub fn verify(
77        &self,
78        ns_table: &NsTable,
79        commit: &VidCommitment,
80        common: &ADVZCommon,
81    ) -> Option<(Vec<Transaction>, NamespaceId)> {
82        match commit {
83            VidCommitment::V0(commit) => {
84                ADVZScheme::is_consistent(commit, common).ok()?;
85                if !ns_table.in_bounds(&self.ns_index) {
86                    return None; // error: index out of bounds
87                }
88
89                let range = ns_table
90                    .ns_range(&self.ns_index, &PayloadByteLen::from_vid_common(common))
91                    .as_block_range();
92
93                match (&self.ns_proof, range.is_empty()) {
94                    (Some(proof), false) => {
95                        // TODO advz_scheme() arg should be u32 to match get_num_storage_nodes
96                        // https://github.com/EspressoSystems/HotShot/issues/3298
97                        let vid = advz_scheme(
98                            ADVZScheme::get_num_storage_nodes(common).try_into().ok()?, // error: failure to convert u32 to usize
99                        );
100
101                        vid.payload_verify(
102                            Statement {
103                                payload_subslice: self.ns_payload.as_bytes_slice(),
104                                range,
105                                commit,
106                                common,
107                            },
108                            proof,
109                        )
110                        .ok()? // error: internal to payload_verify()
111                        .ok()?; // verification failure
112                    },
113                    (None, true) => {}, // 0-length namespace, nothing to verify
114                    (None, false) => {
115                        tracing::error!(
116                            "ns verify: missing proof for nonempty ns payload range {:?}",
117                            range
118                        );
119                        return None;
120                    },
121                    (Some(_), true) => {
122                        tracing::error!("ns verify: unexpected proof for empty ns payload range");
123                        return None;
124                    },
125                }
126
127                // verification succeeded, return some data
128                let ns_id = ns_table.read_ns_id_unchecked(&self.ns_index);
129                Some((self.ns_payload.export_all_txs(&ns_id), ns_id))
130            },
131            VidCommitment::V1(_) => None,
132        }
133    }
134
135    /// Return all transactions in the namespace whose payload is proven by
136    /// `self`. The namespace ID for each returned [`Transaction`] is set to
137    /// `ns_id`.
138    ///
139    /// # Design warning
140    ///
141    /// This method relies on a promise that a [`NsProof`] stores the entire
142    /// namespace payload. If in the future we wish to remove the payload from a
143    /// [`NsProof`] then this method can no longer be supported.
144    ///
145    /// In that case, use the following a workaround:
146    /// - Given a [`NamespaceId`], get a [`NsIndex`] `i` via
147    ///   [`NsTable::find_ns_id`].
148    /// - Use `i` to get a
149    ///   [`NsPayload`](crate::block::namespace_payload::NsPayload) `p` via
150    ///   [`Payload::ns_payload`].
151    /// - Use `p` to get the desired [`Vec<Transaction>`] via
152    ///   [`NsPayload::export_all_txs`](crate::block::namespace_payload::NsPayload::export_all_txs).
153    ///
154    /// This workaround duplicates the work done in [`NsProof::new`]. If you
155    /// don't like that then you could instead hack [`NsProof::new`] to return a
156    /// pair `(NsProof, Vec<Transaction>)`.
157    pub fn export_all_txs(&self, ns_id: &NamespaceId) -> Vec<Transaction> {
158        self.ns_payload.export_all_txs(ns_id)
159    }
